cmake_minimum_required(VERSION 3.21)
project(TingXianEyeProvider LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /EHsc /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
add_library(tingxian-eye-session STATIC session.cpp)
target_link_libraries(tingxian-eye-session PUBLIC Threads::Threads)
target_include_directories(tingxian-eye-session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(tingxian-eye-session-test session_test.cpp)
target_link_libraries(tingxian-eye-session-test PRIVATE tingxian-eye-session)
enable_testing()
add_test(NAME bounded-eye-provider-without-sdk-or-media COMMAND tingxian-eye-session-test)
set_tests_properties(bounded-eye-provider-without-sdk-or-media PROPERTIES TIMEOUT 30)
if(WIN32)
  add_executable(tingxian-eye-worker-fixture worker_main.cpp owner_watch.cpp)
  target_compile_definitions(tingxian-eye-worker-fixture PRIVATE TINGXIAN_EYE_SYNTHETIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(tingxian-eye-worker-fixture PRIVATE tingxian-eye-session)
endif()

# Never built by the ordinary app/installer or public SDK-free CI. No vendor
# binaries/models are linked, copied, downloaded or redistributed by this target.
option(TINGXIAN_AR087_OFFLINE_LAB "Build the explicitly pinned local SDK experiment" OFF)
if(TINGXIAN_AR087_OFFLINE_LAB)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "AR 0.8.7 experiment requires Windows x64")
  endif()
  if(NOT EXISTS "${TINGXIAN_AR087_SOURCE}/nvar/include/nvAR.h" OR NOT EXISTS "${TINGXIAN_AR087_PINS}/ar087_local_pins.h")
    message(FATAL_ERROR "Provide reviewed 0.8.7 source headers and local compile-time runtime pins")
  endif()
  enable_language(C)
  add_library(tingxian-eye-ar087 STATIC ar087_bridge.c ar087_provider.cpp)
  target_include_directories(tingxian-eye-ar087 PRIVATE "${TINGXIAN_AR087_SOURCE}/nvar/include" "${TINGXIAN_AR087_PINS}")
  target_compile_definitions(tingxian-eye-ar087 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(tingxian-eye-ar087 PUBLIC tingxian-eye-session PRIVATE bcrypt)
  add_executable(tingxian-eye-offline offline_main.cpp)
  target_compile_definitions(tingxian-eye-offline PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(tingxian-eye-offline PRIVATE tingxian-eye-ar087)
  add_executable(tingxian-eye-worker worker_main.cpp owner_watch.cpp)
  target_compile_definitions(tingxian-eye-worker PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(tingxian-eye-worker PRIVATE tingxian-eye-ar087)
endif()
