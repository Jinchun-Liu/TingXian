cmake_minimum_required(VERSION 3.21)
project(TingXianVideoSource LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The video source requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# This optional component has no GPU SDK or external package dependencies.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TingXianVideoSource SHARED source.cpp frame_transport.cpp source.def)
target_compile_definitions(TingXianVideoSource PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(TingXianVideoSource PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(TingXianVideoSource PRIVATE mfplat mf mfuuid mfsensorgroup ole32 propsys advapi32 bcrypt)
add_executable(tingxian-video-source-test source_test.cpp)
target_compile_definitions(tingxian-video-source-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-video-source-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-source-test PRIVATE mfplat mf mfuuid ole32 propsys)
enable_testing()
add_test(NAME media-source-without-device-registration
  COMMAND tingxian-video-source-test $<TARGET_FILE:TingXianVideoSource>)
set_tests_properties(media-source-without-device-registration PROPERTIES TIMEOUT 30)
add_executable(tingxian-video-ipc-test ipc_test.cpp frame_transport.cpp)
target_compile_definitions(tingxian-video-ipc-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-video-ipc-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-ipc-test PRIVATE mfplat mf mfuuid ole32 propsys advapi32 bcrypt)
add_test(NAME cross-process-video-without-device-registration
  COMMAND tingxian-video-ipc-test $<TARGET_FILE:TingXianVideoSource>)
set_tests_properties(cross-process-video-without-device-registration PROPERTIES TIMEOUT 45)
add_library(tingxian-video-session STATIC camera_session.cpp camera_windows.cpp frame_transport.cpp worker_owner.cpp)
target_compile_definitions(tingxian-video-session PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-video-session PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-session PUBLIC mfplat mfuuid ole32 advapi32 bcrypt)
add_executable(tingxian-video-session-test camera_session_test.cpp)
target_compile_options(tingxian-video-session-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-session-test PRIVATE tingxian-video-session)
add_test(NAME session-failures-without-device-registration COMMAND tingxian-video-session-test)
set_tests_properties(session-failures-without-device-registration PROPERTIES TIMEOUT 30)
add_executable(tingxian-video-device-probe camera_probe.cpp)
target_compile_options(tingxian-video-device-probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-device-probe PRIVATE tingxian-video-session)
add_test(NAME read-only-device-prerequisites COMMAND tingxian-video-device-probe --probe)
set_tests_properties(read-only-device-prerequisites PROPERTIES TIMEOUT 15)
add_executable(tingxian-video-registration-test registration_test.cpp)
target_compile_definitions(tingxian-video-registration-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-video-registration-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-registration-test PRIVATE mfplat mfuuid ole32)
add_executable(tingxian-video-device-synthetic-test device_synthetic_test.cpp)
target_compile_options(tingxian-video-device-synthetic-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-device-synthetic-test PRIVATE tingxian-video-session mf mfreadwrite)
# Actual devices are opt-in on the target Windows 11 PC, never an ordinary CTest.
add_test(NAME installed-device-tool-description-only COMMAND tingxian-video-device-synthetic-test --describe)
set_tests_properties(installed-device-tool-description-only PROPERTIES TIMEOUT 10)
add_executable(tingxian-video-worker-owner-test worker_owner_test.cpp)
target_compile_options(tingxian-video-worker-owner-test PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-worker-owner-test PRIVATE tingxian-video-session)
add_test(NAME worker-owner-lifetime-without-device-registration COMMAND tingxian-video-worker-owner-test)
set_tests_properties(worker-owner-lifetime-without-device-registration PROPERTIES TIMEOUT 45)
add_executable(tingxian-video-worker worker_main.cpp)
target_compile_options(tingxian-video-worker PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tingxian-video-worker PRIVATE tingxian-video-session)
add_executable(tingxian-video-worker-fixture worker_fixture.cpp)
target_compile_definitions(tingxian-video-worker-fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-video-worker-fixture PRIVATE /W4 /EHsc /utf-8)
