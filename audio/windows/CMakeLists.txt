cmake_minimum_required(VERSION 3.21)
project(TingXianAudioOutput LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The audio adapter requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(tingxian-audio-output STATIC audio_output.cpp audio_windows.cpp)
target_compile_definitions(tingxian-audio-output PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tingxian-audio-output PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(tingxian-audio-output PUBLIC ole32 propsys cfgmgr32)
add_library(TingXianAudioOutput SHARED audio_bridge.cpp audio_bridge.def)
target_compile_options(TingXianAudioOutput PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(TingXianAudioOutput PRIVATE tingxian-audio-output)
# Separate test-only DLL. Its factory never constructs a WindowsBackend and no
# fixture controls are compiled/exported by the production target above.
add_library(TingXianAudioFixture SHARED audio_bridge.cpp audio_bridge.def audio_fixture.cpp)
target_compile_definitions(TingXianAudioFixture PRIVATE TINGXIAN_AUDIO_FIXTURE)
target_compile_options(TingXianAudioFixture PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(TingXianAudioFixture PRIVATE tingxian-audio-output)
add_executable(tingxian-audio-probe audio_probe.cpp)
target_link_libraries(tingxian-audio-probe PRIVATE tingxian-audio-output)
target_compile_options(tingxian-audio-probe PRIVATE /W4 /WX /EHsc /utf-8)
add_executable(tingxian-audio-output-test audio_output_test.cpp)
target_link_libraries(tingxian-audio-output-test PRIVATE tingxian-audio-output)
target_compile_options(tingxian-audio-output-test PRIVATE /W4 /WX /EHsc /utf-8)
enable_testing()
add_test(NAME audio-output-in-memory COMMAND tingxian-audio-output-test)
add_test(NAME audio-endpoints-read-only COMMAND tingxian-audio-probe --probe)
add_test(NAME audio-unknown-pair-no-stream COMMAND tingxian-audio-probe --reject-unknown)
set_tests_properties(audio-output-in-memory audio-endpoints-read-only audio-unknown-pair-no-stream PROPERTIES TIMEOUT 20)

add_library(tingxian-audio-input STATIC capture_input.cpp capture_windows.cpp)
target_compile_options(tingxian-audio-input PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(tingxian-audio-input PUBLIC tingxian-audio-output)
add_library(TingXianAudioCapture SHARED capture_bridge.cpp capture_bridge.def)
target_compile_options(TingXianAudioCapture PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(TingXianAudioCapture PRIVATE tingxian-audio-input)
add_executable(tingxian-audio-input-test capture_input_test.cpp)
target_compile_options(tingxian-audio-input-test PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(tingxian-audio-input-test PRIVATE tingxian-audio-input)
add_executable(tingxian-audio-input-probe capture_probe.cpp)
target_compile_options(tingxian-audio-input-probe PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(tingxian-audio-input-probe PRIVATE TingXianAudioCapture)
add_test(NAME audio-input-in-memory COMMAND tingxian-audio-input-test)
add_test(NAME audio-input-production-no-stream COMMAND tingxian-audio-input-probe --reject-unknown)
set_tests_properties(audio-input-in-memory audio-input-production-no-stream PROPERTIES TIMEOUT 20)
